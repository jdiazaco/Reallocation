/*
This do-file encompasses all data preparation. This means:
1. Generate a fake-file when necessary
2. Transform the GP-codes into the GP02 (by looking at the product portfolio before and after the change and checking with the concordance)
3. Purges firm-years with missing data, data which does not take legitimate values (GP codes which are to short etc) and other such problems
4. Calculate a revenue price index for each product, each firm and each firm without each single product
5. Calculates the pressure from foreign trade for each product category
6. Builds an average pressure over all products for each individual firm
7. Generates measures of firm proximity (industry codes), which bridge the various GP and WZ-schemes
8. Generates a Capital stock using the perpetual inventory method. The first capital stock is approximated with depreciations
9. Calculates weights for each firm according to the weightening scheme of the kse
10. Calculates deflated revenue, labor and intermediate inputs and other variables for the production function


*/



//----------------------------------\\
//----Transform GPs into GP 2002----\\ 
//**********************************\\
{
if "$Transform_GP" == "YES" {


//generate flags to indicate the GP-system used
if  "$IWH_run" == "YES"  {
use land_b bnr jahr if  jahr >= $start_year_load  & $end_year_load >= jahr using "$savepath\startfile_fake_GP",clear
duplicates drop
}

if "$Fernrechnen" == "YES"  |  "$Wissens_Arbeitspl" == "YES"  {
use bnr jahr land_b if jahr >= $start_year_load  & $end_year_load >= jahr using "$Firmendaten", clear
}

drop if bnr==.


gen flag_GP_1995 = 1 if (jahr <2002 & land_b != 13) | (jahr <2001 & land_b==13)
replace flag_GP_1995 = 0 if flag_GP_1995==.
gen flag_GP_2002 = 1 if (jahr >= 2002 & jahr <2009) | (jahr ==2001 & land_b==13)
replace flag_GP_2002 = 0 if flag_GP_2002==.
gen flag_GP_2009 = 1 if jahr >= 2009
replace flag_GP_2009 = 0 if flag_GP_2009==.

quietly xtset bnr jahr
gen flag_new_GP = 1 if flag_GP_2002 == 1 & l.flag_GP_1995 == 1
replace flag_new_GP = 1 if flag_GP_2009 == 1 & l.flag_GP_2002 == 1
replace flag_new_GP = 0 if flag_new_GP == . 

gen flag_old_GP = 1 if f.flag_GP_2002 == 1 & flag_GP_1995 == 1
replace flag_old_GP = 1 if f.flag_GP_2009 == 1 & flag_GP_2002 == 1
replace flag_old_GP = 0 if flag_old_GP == . 


gen flag_switch_group02 = 1 if (flag_GP_2002 == 1 & l.flag_GP_1995 == 1) | (f.flag_GP_2002 == 1 & flag_GP_1995 == 1)
gen flag_switch_group09 = 1 if (flag_GP_2009 == 1 & l.flag_GP_2002 == 1) | (f.flag_GP_2009 == 1 & flag_GP_2002 == 1)

keep jahr bnr flag_switch_group02 flag_switch_group09 flag_new_GP flag_old_GP flag_GP_1995 flag_GP_2002 flag_GP_2009

tab jahr, su (flag_switch_group02)
tab jahr, su (flag_switch_group09)
tab jahr, su (flag_new_GP)
tab jahr, su (flag_old_GP)

tab jahr, su (flag_GP_1995)
tab jahr, su (flag_GP_2002)
tab jahr, su (flag_GP_2009)

compress jahr bnr flag_switch_group02 flag_switch_group09 flag_new_GP flag_old_GP flag_GP_1995 flag_GP_2002 flag_GP_2009

save "$savepath\GP-flags", replace

drop flag_new_GP flag_old_GP

rename flag_GP_1995 flag_GP95
rename flag_GP_2002 flag_GP02
rename flag_GP_2009 flag_GP09

save "$savepath\GP-flags_program_switch_GPs", replace


//-------------------------------------------------------------------------------------------------------\\
//-------------------------------- constructing the file---------------------------------\\
//-------------------------------------------------------------------------------------------------------\\


if  "$IWH_run" == "YES"  {
use land_b pe_10 pe_13 pe_14 pe_15 pe_16 pe_19 pe_24 pe_21 pe_26 pe_29 pe_20 pe_25 pe_30 bnr unr jahr if jahr >= $start_year_load & jahr <= $end_year_load using "$savepath\startfile_fake_GP",clear
duplicates drop
drop if bnr ==.
}
if "$Wissens_Arbeitspl" == "YES" | "$Fernrechnen" == "YES" {
use unr bnr jahr land_b pe_10 if jahr >= $start_year_load & jahr <= $end_year_load using "$Firmendaten", clear
count
drop if bnr == . 
duplicates drop
merge 1:m jahr unr bnr using "$Produktdaten", keepusing(pe_13 pe_14 pe_15 pe_16 pe_19 pe_24 pe_21 pe_26 pe_29 pe_20 pe_25 pe_30 bnr unr jahr)
keep if jahr >= $start_year_load & jahr <= $end_year_load 
tab _merge
drop if _merge != 3
drop _merge
duplicates drop
drop if bnr ==.
}

tab jahr





**************************************************************
*--------------------Construct final_gp02--------------------*
**************************************************************
{


/*
// Very first step. Reclassify GP09V2 to GP09. This is an umambigous reclassification and we can use official concordance tables to do so.
if "$adjust_GP09V2" == "YES"  {
gen GP09V2 = substr(pe_13,1,9) if jahr >=2012 
gen last_dig_GP09V2 = substr(pe_13,10,1) if jahr >=2012 
merge m:1 GP09V2 using "$savepath\Concordances\Konkordanz GP09V2 to GP09.dta"
drop if _merge ==2
replace GP09 = GP09V2 if _merge ==1 & jahr >=2012   // no concordance, hence here gp09V2 is ten digit already
count if length(pe_13) != 10
replace pe_13 = GP09 + last_dig_GP09V2  if jahr >=2012 
count if length(pe_13) != 10

 
drop GP09 last_dig_GP09V2 GP09V2 _merge
}
*/




start_diagnosis pe_15

switch_GPs pe_13 GP95 GP02 bnr target1
switch_GPs pe_13 GP09 GP02 bnr target2

count if target1 != target2 & target1 !="" & target2 !=""

gen final_gp02 = target1 if target1 !=""
replace final_gp02 = target2 if target2 !="" & final_gp02 ==""

gen marker_GP95 = 1 if substr(final_gp02,1,2) == "95"
replace marker_GP95 = 0 if marker_GP95 != 1

gen marker_GP02 = 1 if substr(final_gp02,1,2) == "02"
replace marker_GP02 = 0 if marker_GP02 != 1

gen marker_GP09 = 1 if substr(final_gp02,1,2) == "09"
replace marker_GP09 = 0 if marker_GP09 != 1

tab jahr, sum(marker_GP95)
tab jahr, sum(marker_GP02)
tab jahr, sum(marker_GP09)

drop marker_GP95 marker_GP02 marker_GP09

//thus far, we have a final_gp02 consisting of the classification (first two digits, either 95,02 or 09)
//abd the 10 digit product codes. Should by a twelve digit number.
//However, potentially, these final_gps still have different measurement units. Therefore, we will now
//pick the measurement unit that captures the largest portion of each final_gp

//the following bracket contains this routine
{


	//first, we save all the different values of the three variables giving measurement units in a global
	//each of these values stands for a distinct measurement unit
	levelsof pe_16 
	global different_measurement_units = "`r(levels)'"
	levelsof pe_21
	global different_measurement_units = "$different_measurement_units `r(levels)'"
	levelsof pe_26
	global different_measurement_units = "$different_measurement_units `r(levels)'"
	
	
	//second, we eliminate all one digit measurement_units from the list (because they are illegitimate
	//and cause problems with the elimination of duplicates further down
	local temporary_list = "$different_measurement_units"
	global different_measurement_units = ""
	foreach entry in `temporary_list' {
	if length("`entry'") >= 2 {
	global different_measurement_units = "$different_measurement_units `entry'"
	}
	}
	
	
	
	//third, we eliminate duplicates from this global (same routine as in many programs)
	//We will also eliminate 98 and 9800, which signify Euros as measurement unit
	local unique_list = "$different_measurement_units"
	foreach argument in `unique_list' {
	local unique_list = subinstr("`unique_list'", "`argument' ", "  ", .)
	local unique_list = subinstr("`unique_list'", " `argument'", "  ", .)
	local unique_list = "`unique_list' `argument'"
	}
	global different_measurement_units = "`unique_list'"
	global different_measurement_units = subinstr("$different_measurement_units", "98", "  ", .)
	global different_measurement_units = subinstr("$different_measurement_units", "9800", "  ", .)
	
	
	//fourth, we generate a marker for each measurement units: It is 1 if this observation has 
	//quantity data in this measurement unit, 0 otherwise
	foreach measurement_unit in $different_measurement_units{
	gen marker_`measurement_unit' = 1 if (pe_16 == `measurement_unit' & pe_19 !=0) | (pe_21 == `measurement_unit' & pe_24 !=0) | (pe_26 == `measurement_unit' & pe_29 !=0)
	replace marker_`measurement_unit' = 0 if marker_`measurement_unit'==.
	}
	
	
	
	
	//in a last step, we pick the measurement unit. This has to be done iteratively: 
	//we first pick the measurement unit which allows us to have the most observations
	//per final_gp02
	//however, among the remaining observations, we have to assign a measurement unit as well. 
	//thus, we will do a while loop that cycles through assignement steps as long as there 
	//are still observations with quantity data but without a measurement unit.
	gen marker_unit_assigned = 0
	gen measurement_unit =""
	local obs_without_unit_but_data = 1
	
	
	while `obs_without_unit_but_data' > 0 {
	
			
			
			*1st, we sort all obs with the same final_gp02 and still missing measurement_unit into a group
			//then we count how often each measurement unit actually occurs within that group
			sort final_gp02 marker_unit_assigned
			foreach measurement_unit in $different_measurement_units {
			by final_gp02 marker_unit_assigned: egen count_`measurement_unit' = total( marker_`measurement_unit') if marker_unit_assigned ==0
			}
	
	
	
			*2nd, we select the measurement unit which most observations have and set it as measurement unit for the entire group.
			egen max_num_of_obs_per_unit = rowmax( count_* )
			foreach measurement_unit in $different_measurement_units {
			replace measurement_unit = "`measurement_unit'" if count_`measurement_unit' == max_num_of_obs_per_unit & marker_`measurement_unit' == 1 & measurement_unit==""
			}
	
	
			*3rd we do some housecleaning to keep the loop going: 
			//tabulate how many measurement units we have, mark observations with a measurement unit
			//count how many observations still have to be found
			replace marker_unit_assigned = 1 if measurement_unit !=""
			tab jahr, su (marker_unit_assigned)
			count if max_num_of_obs_per_unit > 0 & marker_unit_assigned == 0
			local obs_without_unit_but_data = `r(N)'
			drop count_* max_num_of_obs_per_unit
			
			
	}
	
	//if there is no quantity data, replace the variable as missing
	//count the number of empty measurement units
	count if measurement_unit == ""
	//count the number of empty measurement units that also have no quantity data (mus be equal)
	count if measurement_unit == "" & (pe_19 ==.|pe_19 ==0) & (pe_24==.|pe_24==0) & (pe_29 ==.|pe_29==0)
	//count the obs with empty quantity data but a measurement unit (should be 0)
	count if (pe_19 ==.|pe_19 ==0) & (pe_24==.|pe_24==0) & (pe_29 ==.|pe_29==0) & measurement_unit != ""
	replace measurement_unit = "no" if measurement_unit == ""
	
	//drop the helpvars for the loop
	drop marker_*

//close the measurement unit routine
}

//After finding out the measurement unit, attach it to the final_gp
replace final_gp02 = final_gp02 + measurement_unit



//This part gathers information on the wz category in a new variable
gen wz02_4digit = target1_4digit if flag_GP95 == 1
gen wz02_3digit = target1_3digit if flag_GP95 == 1
gen wz02_2digit = target1_2digit if flag_GP95 == 1

replace wz02_4digit = target2_4digit if flag_GP95 != 1
replace wz02_3digit = target2_3digit if flag_GP95 != 1
replace wz02_2digit = target2_2digit if flag_GP95 != 1



//diagnose the severity of the problem of unclear wz-s:
//----------------------------------------------------

	//wz 4 digit
	gen marker_unclear = 1 if wz02_4digit == "unclear"
	replace marker_unclear = 0 if marker_unclear == .
	tab jahr, sum(marker_unclear)
	drop marker_unclear
	
	//wz 3 digit
	gen marker_unclear = 1 if wz02_3digit == "unclear"
	replace marker_unclear = 0 if marker_unclear == .
	tab jahr, sum(marker_unclear)
	drop marker_unclear
	
	//wz 2 digit
	gen marker_unclear = 1 if wz02_2digit == "unclear"
	replace marker_unclear = 0 if marker_unclear == .
	tab jahr, sum(marker_unclear)
	drop marker_unclear
	
	
drop target*
}

**************************************************************
*---------Construct final_gp09 (and industry info)-----------*
**************************************************************
{

set more off
switch_GPs final_gp02 GP02 GP09 bnr target1
switch_GPs final_gp02 GP95 GP09 bnr target2

count if target1 != target2 & target1 !="" & target2 !="" & jahr < 2009
count if measurement_unit ==""
gen final_gp09 = "09" + pe_13 + measurement_unit if jahr > 2008
replace final_gp09 = target1 + measurement_unit if target1 !="" 
replace final_gp09 = target2 + measurement_unit if target2 !="" 

gen wz08_4digit = substr(final_gp09, 3, 4) if jahr > 2008
gen wz08_3digit = substr(final_gp09, 3, 3) if jahr > 2008
gen wz08_2digit = substr(final_gp09, 3, 2) if jahr > 2008

replace wz08_4digit = target1_4digit if flag_GP02 == 1 & wz08_4digit==""
replace wz08_3digit = target1_3digit if flag_GP02 == 1 & wz08_3digit==""
replace wz08_2digit = target1_2digit if flag_GP02 == 1 & wz08_2digit==""

replace wz08_4digit = target2_4digit if flag_GP95  == 1 & wz08_4digit==""
replace wz08_3digit = target2_3digit if flag_GP95  == 1 & wz08_3digit==""
replace wz08_2digit = target2_2digit if flag_GP95  == 1 & wz08_2digit==""


//diagnose the severity of the problem of unclear wz-s:
//----------------------------------------------------

	//wz 4 digit
	gen marker_unclear = 1 if wz08_4digit == "unclear"
	replace marker_unclear = 0 if marker_unclear == .
	tab jahr, sum(marker_unclear)
	drop marker_unclear
	
	//wz 3 digit
	gen marker_unclear = 1 if wz08_3digit == "unclear"
	replace marker_unclear = 0 if marker_unclear == .
	tab jahr, sum(marker_unclear)
	drop marker_unclear
	
	//wz 2 digit
	gen marker_unclear = 1 if wz08_2digit == "unclear"
	replace marker_unclear = 0 if marker_unclear == .
	tab jahr, sum(marker_unclear)
	drop marker_unclear


//tabulate the success in assigning a wz2 per year and switch step
gen marker_success = 1 if wz08_2digit != "unclear"
replace marker_success = 0 if marker_success ==.

tab jahr, su (marker_success)
su marker_success if flag_GP95 ==1
su marker_success if flag_GP02 ==1

drop marker_success

drop target*
}
//


**************************************************************
*--------gen p & q & market sizes, clean up and save---------*
**************************************************************
//merge with data on which pe_13 is reported in which classification

	//dropping the variables with the GP-flags (since they can be from final_gp or from original data
	//this ambiguity makes it unclear what they are really saying, which depends on the last switch_GP command
	foreach necessary_vars in flag_switch_group02 flag_switch_group09 flag_GP95 flag_GP02 flag_GP09 {
		capture drop `necessary_vars'
		}
	merge m:1 bnr jahr using "$savepath\GP-flags"
	drop if _merge == 2
	drop _merge

//generate Q and p
	tostring (pe_16), replace
	tostring (pe_21), replace
	tostring (pe_26), replace
	gen Q = pe_19 if pe_16 == measurement_unit & pe_16 !=""
	replace Q = pe_24 if pe_21 == measurement_unit & pe_21 !=""
	replace Q = pe_29 if pe_26 == measurement_unit & pe_26 !=""
	
	gen WVquant = pe_20 if pe_16 == measurement_unit & pe_16 !=""
	replace WVquant = pe_25 if pe_21 == measurement_unit & pe_21 !=""
	replace WVquant = pe_30 if pe_26 == measurement_unit & pe_26 !=""

	
//control whether there are quantities without measurement units or vice versa (expect two "0")
	count if measurement_unit !="no" & (Q ==. | Q ==0)
	count if measurement_unit =="no" & (Q !=. & Q !=0)
	drop pe_16 pe_19 pe_21 pe_24 pe_26 pe_29
	gen p = pe_15/Q
	gen mark_price = 1 if p !=.
	replace mark_price = 0 if mark_price != 1
	

	
////////////////////////////
	///////////we saved a test-file here
////////////////////////////

	
	
	
// drop firm with unbelievably high price deviations from the industry average
// we assume that this is measurement error. We know, e.g., that some firms have a jump of 1000 => they go from kg to tons though it's not allowed
	// IMPORTANT WE ADD MEASUREMENT UNIT TO IDENTIFIER AS SAME PE_13 HAS SOMETIMES DIFFERENT MEASUREMENTUNIT
	// WE ALSO NEED A STRING FOR THIS TO CORRECTLY AGGREGATE PRICES BY THIS STRING
	
	
	gen ident_pe13_measure_unit =  pe_13 +  measurement_unit
	egen help_id = group (ident_pe13_measure_unit bnr  flag_GP_1995 flag_GP_2002 flag_GP_2009)   // before without measurement_unit
	xtset help_id jahr
	
	
	
	gen price_growth = p/l.p
	bys ident_pe13_measure_unit  flag_GP_1995 flag_GP_2002 flag_GP_2009: egen industry_price = mean(p)
	by ident_pe13_measure_unit  flag_GP_1995 flag_GP_2002 flag_GP_2009: egen N = total(mark_price)
	drop mark_price
	
	
	//test in which we eliminate observations that greatly influence the aggregate index => extreme values
	gen industry_price_without_obs = (industry_price - 1/N * p) * N/(N-1) if N > 1
	gen test_influence = p / industry_price_without_obs

	// decide how much prices are dropped due to missreporting
	if "$less_price_dropping" != "YES" {
	_pctile test_influence , nq(100)
	gen OK_test_influence = inrange(test_influence, `r(r1)', `r(r99)') if . != test_influence
	}
	
	if "$less_price_dropping" == "YES" {
	_pctile test_influence , nq(1000)
	gen OK_test_influence = inrange(test_influence, `r(r1)', `r(r999)') if . != test_influence
	}

	
	// summarize and show implausible high derivation from product price mean 
	su test_influence
	cap noisily kdensity test_influence if OK_test_influence == 1
	cap noisily graph export "$savepath\Freigabe\Descriptive\Ratio_Price_to_avg_product_price_valid_obs.emf", replace
	su test_influence if OK_test_influence == 1 , det
	cap noisily kdensity test_influence if OK_test_influence == 0
	cap noisily graph export "$savepath\Freigabe\Descriptive\Ratio_Price_to_avg_product_price_invalid_obs.emf", replace
	su test_influence if OK_test_influence == 0 , det
	
	gen marker_reporting_error = 1 if OK_test_influence == 0 //choose between OK_test_influence marker_extreme

	replace p = . if marker_reporting_error == 1
	replace Q = . if marker_reporting_error == 1
	
	su test_influence if marker_reporting_error != 1
	kdensity test_influence if marker_reporting_error != 1
	
	drop help_id price_growth industry_price N industry_price_without_obs test_influence marker_reporting_error
	
	
	

	
//count whether there are prices without measurement units or vice versa (expect two "0")
	count if p ==. & measurement_unit !="no"
	count if p !=. & measurement_unit =="no"
	replace measurement_unit = "no" if p ==. & measurement_unit !="no"



//to compute the market shares correctly in the firm price computation, we should only use observations were there is a price
//hence, we create a flag for this
	gen flag_quantity_exists = 1 if p != . & Q != . 
	replace flag_quantity_exists = 0 if flag_quantity_exists == .
	

//
//generate market sizes for use further down

	bys ident_pe13_measure_unit jahr: egen market_size_GP10 = total(pe_15)
	by ident_pe13_measure_unit jahr : egen market_size_GP10_for_price = total(pe_15 * flag_quantity_exists)
	
	
	bys ident_pe13_measure_unit unr jahr : egen Q_GP10 = total(Q) 
	bys ident_pe13_measure_unit bnr jahr : egen Q_GP10_bnr = total(Q) 
	
	bys ident_pe13_measure_unit jahr : egen Market_size_Q =  total(Q)


	
//generate firm-product specific market shares and prices
	bys ident_pe13_measure_unit unr jahr: egen AbsatzGP10 = total(pe_15)
	by ident_pe13_measure_unit unr jahr : egen AbsatzGP10_if_quantity_exists = total(pe_15* flag_quantity_exists) 
	
	bys ident_pe13_measure_unit bnr jahr: egen AbsatzGP10_bnr = total(pe_15)
	by ident_pe13_measure_unit bnr jahr : egen AbsatzGP10_if_quantity_exists_b = total(pe_15* flag_quantity_exists) 
	
	
	gen P_com = ((AbsatzGP10_if_quantity_exists/Q_GP10)/(market_size_GP10_for_price/Market_size_Q))
	gen P_P = (AbsatzGP10_if_quantity_exists/Q_GP10)
	
	
	
	gen P_com_bnr = ((AbsatzGP10_if_quantity_exists_b/Q_GP10_bnr)/(market_size_GP10_for_price/Market_size_Q))
	gen P_P_bnr = (AbsatzGP10_if_quantity_exists_b/Q_GP10_bnr)
	
	
	
// define a new revenue weight
gen market_price = market_size_GP10_for_price/Market_size_Q
gen market_price_revenue =  Q*market_price

gen TEST_diff_rev_market_rev = pe_15/market_price_revenue if  flag_quantity_exists == 1

su TEST_diff_rev_market_rev, det
kdensity TEST_diff_rev_market_rev 
graph export "$savepath\Freigabe\Priceindices\kdensity_TEST_diff_rev_market_rev.pdf", replace
drop TEST_diff_rev_market_rev market_price




 
	
//generate market shares for each pe_13
	gen ms_p = AbsatzGP10 / market_size_GP10
	gen ms_pq = Q_GP10 / Market_size_Q

	gen ms_p_bnr = AbsatzGP10_bnr / market_size_GP10
	gen ms_pq_bnr = Q_GP10_bnr / Market_size_Q
	
	

bys unr jahr: egen Absatzproduktionswert_unr = total(pe_15)
by unr jahr: egen Absatzprod_unr_Quan = total(pe_15 * flag_quantity_exists)
by unr jahr: egen Absatzprod_unr_Quan_market = total(market_price_revenue * flag_quantity_exists)
	
	// gen variables of market shares 
by unr jahr: egen ms_f = total(ms_p * pe_15/Absatzproduktionswert_unr)
 
by unr jahr: egen ms_Qf = total(ms_pq * flag_quantity_exists * pe_15/Absatzprod_unr_Quan)

// gen more additional control / state  variables for productivity 
// Output firm price
by unr jahr: egen FirmP = total(P_com * flag_quantity_exists * pe_15/Absatzprod_unr_Quan)
by unr jahr: egen Firmp = total(P_P * flag_quantity_exists * pe_15/Absatzprod_unr_Quan)

by unr jahr: egen FirmP_market = total(P_com * flag_quantity_exists * market_price_revenue/Absatzprod_unr_Quan_market)
by unr jahr: egen Firmp_market = total(P_P * flag_quantity_exists * market_price_revenue/Absatzprod_unr_Quan_market)



//bnr market shares etc  

bys bnr jahr: egen Absatzproduktionswert_bnr = total(pe_15)
by bnr jahr: egen Absatzprod_bnr_Quan = total(pe_15 * flag_quantity_exists)
by bnr jahr: egen Absatzprod_bnr_Quan_market = total(market_price_revenue * flag_quantity_exists)
	
	// gen variables of market shares 
by bnr jahr: egen ms_rev_bnr = total(ms_p_bnr * pe_15/Absatzproduktionswert_bnr)
 
by bnr jahr: egen ms_q_bnr = total(ms_pq_bnr * flag_quantity_exists * pe_15/Absatzprod_bnr_Quan)

// gen more additional control / state  variables for productivity 
// Output firm price
by bnr jahr: egen PlantP = total(P_com_bnr * flag_quantity_exists * pe_15/Absatzprod_bnr_Quan)
by bnr jahr: egen Plantp = total(P_P_bnr * flag_quantity_exists * pe_15/Absatzprod_bnr_Quan)


	
	gen ms_p_within = AbsatzGP10 / Absatzproduktionswert_unr


	gen ms_p_bnr_within = AbsatzGP10_bnr / Absatzproduktionswert_bnr

	




replace PlantP = . if PlantP == 0	
replace Plantp = . if Plantp == 0	

replace FirmP = . if FirmP == 0	
replace Firmp = . if Firmp == 0	
replace FirmP_market = . if FirmP_market == 0	
replace Firmp_market = . if Firmp_market == 0	
// su prices p com and p p 	
su P_com, det
su P_P, det	
	
// su firm prices BEFORE imputation	
su FirmP, det
su Firmp, det

su PlantP, det
su Plantp, det


su FirmP_market, det
su Firmp_market, det
